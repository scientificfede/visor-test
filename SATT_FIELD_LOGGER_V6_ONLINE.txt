function calculateTunaWeightKg(Lval, Gval){
  const L = numberFromInput(Lval);
  const G = numberFromInput(Gval);
  if(!(L > 0) || !(G > 0)) return '';
  const kg = ((((0.9442 * L + 2.9475) / 2.54) * ((G * 2) / 2.54) * ((G * 2) / 2.54)) / 800) * 0.454;
  return Number.isFinite(kg) ? (Math.round(kg * 10) / 10).toFixed(1) : '';
}

function parseCoord(input, axis){
  const original = String(input || '').trim();
  if(!original) return {ok:false, value:null, msg:'sin coordenada'};
  let s = original.toUpperCase().replace(/,/g,'.');
  const hemiMatch = s.match(/[NSEW]/);
  const hemi = hemiMatch ? hemiMatch[0] : '';
  let sign = /^\s*-/.test(s) ? -1 : 1;
  if(hemi === 'S' || hemi === 'W') sign = -1;
  if(hemi === 'N' || hemi === 'E') sign = 1;
  s = s.replace(/[NSEW]/g,' ');
  const nums = s.match(/-?\d+(?:\.\d+)?/g);
  if(!nums || nums.length === 0) return {ok:false, value:null, msg:'formato no reconocido'};
  let deg = Math.abs(parseFloat(nums[0]));
  let min = nums.length > 1 ? Math.abs(parseFloat(nums[1])) : 0;
  let sec = nums.length > 2 ? Math.abs(parseFloat(nums[2])) : 0;
  if(nums.length > 3) return {ok:false,value:null,msg:'demasiados números'};
  if(min >= 60 || sec >= 60) return {ok:false,value:null,msg:'minutos/segundos fuera de rango'};
  let dec = sign * (deg + min/60 + sec/3600);
  const limit = axis === 'lat' ? 90 : 180;
  if(Math.abs(dec) > limit) return {ok:false,value:null,msg:`fuera de rango (${axis === 'lat' ? 'latitud' : 'longitud'})`};
  return {ok:true, value:Number(dec.toFixed(7)), msg:''};
}

  return `Coordenada pendiente o con error: lat ${lat.ok ? lat.value : lat.msg}; lon ${lon.ok ? lon.value : lon.msg}`;
}

function startGPSWatch(){
  if(!('geolocation' in navigator)){
    app.markVessel.gpsError = 'Este navegador no permite geolocalización.';
    app.markVessel.gpsActive = false;
    saveApp(); renderAll();
    alert(app.markVessel.gpsError);
    return;
  }
  if(gpsWatchId !== null) navigator.geolocation.clearWatch(gpsWatchId);
  app.markVessel.gpsActive = true;
  app.markVessel.gpsError = '';
  saveApp(); renderAll();
  gpsWatchId = navigator.geolocation.watchPosition(pos => {
    const c = pos.coords;
    app.markVessel.name = 'Embarcación de marcaje';
    app.markVessel.lastLat = Number(c.latitude.toFixed(7));
    app.markVessel.lastLon = Number(c.longitude.toFixed(7));
    app.markVessel.accuracy = c.accuracy != null ? Math.round(c.accuracy) : null;
    app.markVessel.altitude = c.altitude != null ? Math.round(c.altitude) : null;
    app.markVessel.speed = c.speed != null ? Number((c.speed * 1.94384).toFixed(1)) : null; // nudos
    app.markVessel.heading = c.heading != null && !Number.isNaN(c.heading) ? Math.round(c.heading) : null;
    app.markVessel.updatedAt = nowISO();
    app.markVessel.gpsActive = true;
    app.markVessel.gpsError = '';
    saveApp();
    const now = Date.now();
    if(now - lastGpsRender > 900){ lastGpsRender = now; renderAll(); }
  }, err => {
    app.markVessel.gpsActive = false;
    app.markVessel.gpsError = gpsErrorMessage(err);
    saveApp(); renderAll();
  }, { enableHighAccuracy:true, maximumAge:2000, timeout:15000 });
}
function stopGPSWatch(){
  if(gpsWatchId !== null) navigator.geolocation.clearWatch(gpsWatchId);
  gpsWatchId = null;
  app.markVessel.gpsActive = false;

}

function toRad(d){ return d * Math.PI / 180; }
function toDeg(r){ return r * 180 / Math.PI; }
function normalizeBearing(d){ return ((d % 360) + 360) % 360; }
function distanceBearing(lat1, lon1, lat2, lon2){
  const R = 6371.0088; // km
  const phi1=toRad(lat1), phi2=toRad(lat2), dphi=toRad(lat2-lat1), dlambda=toRad(lon2-lon1);
  const a=Math.sin(dphi/2)**2 + Math.cos(phi1)*Math.cos(phi2)*Math.sin(dlambda/2)**2;
  const c=2*Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
  const km=R*c;
  const y=Math.sin(dlambda)*Math.cos(phi2);
  const x=Math.cos(phi1)*Math.sin(phi2) - Math.sin(phi1)*Math.cos(phi2)*Math.cos(dlambda);
  const brg=normalizeBearing(toDeg(Math.atan2(y,x)));
  return {km, nm: km/1.852, bearing: brg};
}
function bearingText(deg){
  if(deg == null || Number.isNaN(deg)) return '';
  const dirs=['N','NNE','NE','ENE','E','ESE','SE','SSE','S','SSW','SW','WSW','W','WNW','NW','NNW'];
  return dirs[Math.round(deg/22.5)%16];
}
function bearingArrow(deg){
  if(deg == null || Number.isNaN(deg)) return '·';
  const arrows=['↑','↗','→','↘','↓','↙','←','↖'];
  return arrows[Math.round(deg/45)%8];
}

function fmtShort(iso){
  if(!iso) return '';
  try{ const d=new Date(iso); return d.toLocaleString([], {day:'2-digit',month:'2-digit',hour:'2-digit',minute:'2-digit',second:'2-digit'}); }catch(e){return iso;}
}
function isoToLocalInput(iso){
  if(!iso) return '';
  const d = new Date(iso); if(isNaN(d)) return '';
  const local = new Date(d.getTime() - d.getTimezoneOffset()*60000);
  return local.toISOString().slice(0,16);
}
function localInputToISO(v){
  if(!v) return '';
  const d = new Date(v); if(isNaN(d)) return '';
  return d.toISOString();
}
function durationText(startISO, endISO){
  if(!startISO) return '';
  const start = new Date(startISO); const end = endISO ? new Date(endISO) : new Date();
  if(isNaN(start) || isNaN(end)) return '';
  let sec = Math.max(0, Math.floor((end-start)/1000));
  const h=Math.floor(sec/3600); sec-=h*3600; const m=Math.floor(sec/60); const s=sec-m*60;
  return `${pad(h)}:${pad(m)}:${pad(s)}`;
}
function fightToEmbarkDuration(e){
  if(!e || !e.biteTime) return '';
  if(e.embarkedTime) return durationText(e.biteTime, e.embarkedTime);
  if(e.status === 'lost' && e.lostTime) return durationText(e.biteTime, e.lostTime);
  if(e.status === 'active') return durationText(e.biteTime);
  return '';
}
function handlingDuration(e){
  if(!e || !e.embarkedTime) return '';
  if(e.markedTime) return durationText(e.embarkedTime, e.markedTime);
  if(e.status === 'embarked') return durationText(e.embarkedTime);
  return '';
}
function totalDuration(e){
  if(!e || !e.biteTime) return '';
  return durationText(e.biteTime, e.markedTime || e.lostTime || (isOpenStatus(e.status) ? nowISO() : ''));
}

  for(let n=0;n<256;n++){ c=n; for(let k=0;k<8;k++){ c = ((c&1) ? (0xEDB88320 ^ (c>>>1)) : (c>>>1)); } table[n]=c>>>0; }

}
const CRC_TABLE = makeCRCTable();
function crc32(bytes){
  let crc = 0 ^ (-1);
  for(let i=0;i<bytes.length;i++){ crc = (crc>>>8) ^ CRC_TABLE[(crc ^ bytes[i]) & 0xFF]; }
  return (crc ^ (-1)) >>> 0;
}
function dosDateTime(date){

  const time = (d.getHours()<<11) | (d.getMinutes()<<5) | Math.floor(d.getSeconds()/2);

  const dosdate = (year<<9) | (month<<5) | day;
  return {time, date:dosdate};
}
